def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def extract_test_patch(eval_path: Path, destination: Path) -> None:
    text = eval_path.read_text(encoding="utf-8", errors="replace").replace("\r\n", "\n").replace("\r", "\n")
    match = re.search(r"git apply -v - <<'(?P<marker>[^']+)'\n(?P<patch>.*?)\n(?P=marker)", text, flags=re.DOTALL)
    if match is None:
        raise RuntimeError(f"test_patch_not_found:{eval_path}")
    destination.write_text(match.group("patch") + "\n", encoding="utf-8", newline="\n")


def filter_patch(source: Path, destination: Path, allowed_paths: list[str]) -> tuple[str, str]:
    original = source.read_text(encoding="utf-8", errors="replace").replace("\r\n", "\n").replace("\r", "\n")
    blocks: list[str] = []
    current: list[str] = []
    for line in original.splitlines(keepends=True):
        if line.startswith("diff --git ") and current:
            blocks.append("".join(current))
            current = []
        current.append(line)
    if current:
        blocks.append("".join(current))
    selected = [block for block in blocks if any(f"a/{path}" in block.splitlines()[0] and f"b/{path}" in block.splitlines()[0] for path in allowed_paths)]
    if not selected:
        raise RuntimeError(f"allowed_patch_block_not_found:{source}:{allowed_paths}")
    destination.write_text("".join(selected), encoding="utf-8", newline="\n")
    return sha256(source), sha256(destination)


def parse_counts(output: str, runner: str) -> tuple[int, int, int, bool]:
    if runner == "pytest":
        passed_match = re.search(r"(\d+) passed", output)
        failed_match = re.search(r"(\d+) failed", output)
        skipped_match = re.search(r"(\d+) skipped", output)
        passed = int(passed_match.group(1)) if passed_match else 0
        failed = int(failed_match.group(1)) if failed_match else 0
        skipped = int(skipped_match.group(1)) if skipped_match else 0
        return passed, failed, skipped, any(match is not None for match in (passed_match, failed_match, skipped_match))
    finished_match = re.search(r"tests finished: (\d+) passed(?:, (\d+) failed)?(?:, (\d+) skipped)?", output)
    if finished_match:
        passed = int(finished_match.group(1))
        failed = int(finished_match.group(2) or 0)
        skipped = int(finished_match.group(3) or 0)
        return passed, failed, skipped, True
    ran_match = re.search(r"Ran (\d+) tests? in ", output)
    ran = int(ran_match.group(1)) if ran_match else 0
    ok = re.search(r"(?:^|\n)OK(?:\s|$)", output) is not None
    failed = re.search(r"(?:^|\n)(?:FAILED|FAIL)(?:\s|$)", output) is not None or "FAILED (" in output
    passed = ran if ok and not failed else 0
    return passed, ran - passed, 0, ran_match is not None


def run_case(task: dict[str, object], method: str, solution_patch: Path, test_patch: Path) -> dict[str, object]:
    task_id = str(task["task_id"])
    log_path = OUT / f"{task_id}.{method}.docker.log"
    solution_step = "" if method == "baseline" else "git apply --ignore-whitespace --verbose /tmp/solution.diff; "
    install_step = f"{task['install']}; " if task["install"] else ""
    shell = "".join([
        "set -u; cd /testbed; git config --global --add safe.directory /testbed; ",
        f"git reset --hard {task['base_commit']}; ",
        "printf 'HEAD='; git rev-parse HEAD; printf 'DIRTY='; git status --porcelain | wc -l; ",
        solution_step,
        "git apply --ignore-whitespace --verbose /tmp/test.diff; ",
        "source /opt/miniconda3/bin/activate; conda activate testbed; ",
        install_step,
        str(task["test_command"]),
    ])
    command = [
        "docker", "run", "--rm", "--user", "root", "--entrypoint", "/bin/bash",
        "--mount", f"type=bind,source={solution_patch},target=/tmp/solution.diff,readonly",
        "--mount", f"type=bind,source={test_patch},target=/tmp/test.diff,readonly",
        str(task["image"]), "-lc", shell,
    ]
    completed = subprocess.run(command, cwd=ROOT, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, encoding="utf-8", errors="replace")
    log_path.write_text(completed.stdout, encoding="utf-8")
    passed, failed, skipped, executed = parse_counts(completed.stdout, str(task["runner"]))
    patch_count = completed.stdout.count("Applied patch")
    solution_applied = method != "baseline" and patch_count >= 2
    test_applied = patch_count >= 1
    reset_ok = f"HEAD={task['base_commit']}" in completed.stdout and "DIRTY=0" in completed.stdout
    return {
        "task_id": task_id,
        "method": method,
        "repository": task["repository"],
        "image": task["image"],
        "base_commit": task["base_commit"],
        "test_file": task["test_file"],
        "solution_patch_sha256": sha256(solution_patch),
        "test_patch_sha256": sha256(test_patch),
        "reset_ok": reset_ok,
        "solution_patch_applied": solution_applied,
        "test_patch_applied": test_applied,
        "test_executed": executed,
        "tests_passed": passed,
        "tests_failed": failed,
        "tests_skipped": skipped,
        "tests_run": passed + failed + skipped,
        "returncode": completed.returncode,
        "resolved_process_signal": bool(method != "baseline" and reset_ok and solution_applied and test_applied and executed and failed == 0 and completed.returncode == 0),
        "formal_outcome": False,
        "provider_calls_added": 0,
    }


def main() -> int:
    if OUT.exists() or ZIP.exists():
        raise SystemExit(f"refusing_to_overwrite:{OUT}")
    if sha256(OBJECTIVE) != ACTIVE_SHA:
        raise SystemExit("active_objective_sha_mismatch")
    OUT.mkdir(parents=True)
    all_results: list[dict[str, object]] = []
    task_provenance: list[dict[str, object]] = []
    for task in TASKS:
        source_dir = BASE_DIR / str(task["relative_dir"])
        gold_source = source_dir / "patch.diff"
        eval_path = source_dir / "eval.sh"
        orca_source = ORCA_ROOT / str(task["orca_task_path"]) / "patch.diff"
        test_patch = OUT / f"{task['task_id']}.test.patch"
        gold_patch = OUT / f"{task['task_id']}.gold.patch"
        orca_patch = OUT / f"{task['task_id']}.orcaloca.patch"
        extract_test_patch(eval_path, test_patch)
        gold_patch.write_text(gold_source.read_text(encoding="utf-8", errors="replace").replace("\r\n", "\n").replace("\r", "\n"), encoding="utf-8", newline="\n")
        original_orca_sha, filtered_orca_sha = filter_patch(orca_source, orca_patch, list(task["allowed_paths"]))
        task_provenance.append({
            "task_id": task["task_id"],
            "gold_source": str(gold_source),
            "gold_patch_sha256": sha256(gold_patch),
            "test_source": str(eval_path),
            "test_patch_sha256": sha256(test_patch),
            "orcaloca_source": str(orca_source),
            "orcaloca_source_sha256": original_orca_sha,
            "orcaloca_filtered_sha256": filtered_orca_sha,
            "orcaloca_allowed_paths": task["allowed_paths"],
            "base_commit": task["base_commit"],
            "image": task["image"],
        })
        baseline_patch = OUT / f"{task['task_id']}.baseline.patch"
        baseline_patch.write_text("", encoding="utf-8", newline="\n")
        all_results.extend([
            run_case(task, "baseline", baseline_patch, test_patch),
            run_case(task, "gold", gold_patch, test_patch),
            run_case(task, "orcaloca", orca_patch, test_patch),
        ])
    fields = list(dict.fromkeys(key for row in all_results for key in row))
    with (OUT / "result_rows.csv").open("w", encoding="utf-8", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=fields)
        writer.writeheader()
        writer.writerows(all_results)
    summary = {
        "analysis_id": OUT.name,
        "generated_at_utc": datetime.now(timezone.utc).isoformat(),
        "active_objective_sha256": ACTIVE_SHA,
        "task_count": len(TASKS),
        "result_rows": len(all_results),
        "test_executed_rows": sum(bool(row["test_executed"]) for row in all_results),
        "resolved_process_signal_rows": sum(bool(row["resolved_process_signal"]) for row in all_results),
        "formal_outcome_rows": 0,
        "provider_calls_added": 0,
        "formal_promotion": False,
        "status": "completed_cached_two_task_three_condition_replay" if all(bool(row["test_executed"]) for row in all_results) else "partial_cached_two_task_three_condition_replay",
        "claim_boundary": "Cached exact-base Linux container process/test diagnostics for Matplotlib-23299 and SymPy-11400. F2P is a test-sensitivity diagnostic, not semantic, robust, causal, survival, or formal outcome.",
        "task_provenance": task_provenance,
        "results": all_results,
    }
    (OUT / "summary.json").write_text(json.dumps(summary, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    (OUT / "RESULTS_ZH.md").write_text(
        "# Matplotlib-23299 / SymPy-11400 cached exact-base three-condition replay\n\n"
        "该包记录两个缓存 exact-base 镜像的 baseline/gold/OrcaLoca 三条件 process/test 诊断；OrcaLoca patch 按任务路径过滤，原始与过滤后 hash 均保留。\n\n"
        "不产生 provider 调用或 formal outcome。\n",
        encoding="utf-8",
    )
    files = sorted(path for path in OUT.rglob("*") if path.is_file())
    (OUT / "file_manifest.json").write_text(json.dumps([{"path": str(path.relative_to(OUT)).replace("\\", "/"), "bytes": path.stat().st_size, "sha256": sha256(path)} for path in files], ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    files = sorted(path for path in OUT.rglob("*") if path.is_file())
    with zipfile.ZipFile(ZIP, "w", compression=zipfile.ZIP_DEFLATED) as archive:
        for path in files:
            archive.write(path, path.relative_to(OUT))
    with zipfile.ZipFile(ZIP) as archive:
        if archive.testzip() is not None:
            raise SystemExit("archive_integrity_failed")
    print(json.dumps(summary, ensure_ascii=False))
    return 0 if summary["status"] == "completed_cached_two_task_three_condition_replay" else 1


if __name__ == "__main__":
    raise SystemExit(main())
