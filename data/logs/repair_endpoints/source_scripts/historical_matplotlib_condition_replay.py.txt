from pathlib import Path


ROOT = Path(__file__).resolve().parents[1]
OBJECTIVE = Path(r"<USER_HOME>\.codex\attachments\8d2f80cc-875f-48d1-8eff-7a500c2d87c5\goal-objective.md")
ACTIVE_SHA = "390f6f7e27993fe039dc2fc278d840b6870ac939c5bacca3232960f9b7e273e9"
OUT = ROOT / "artifacts/work2_matplotlib23476_three_condition_replay_20260907_v2"
ZIP = OUT.with_suffix(".zip")
IMAGE = "swebench/sweb.eval.x86_64.matplotlib_1776_matplotlib-23476:latest"
IMAGE_ID = "sha256:10230151c40fbb30c572497c0d2cc6bebead75042568f9223c3266043e6fc249"
TASK_ID = "matplotlib__matplotlib-23476"
BASE_COMMIT = "33a0599711d26dc2b79f851c6daed4947df7c167"
GOLD_PATCH = ROOT / "artifacts/work2_t0_matplotlib23476_docker_replay_20260906_v1/solution_source_patch.diff"
TEST_PATCH = ROOT / "artifacts/work2_t0_matplotlib23476_docker_replay_20260906_v1/evaluation_test_patch_adapted.diff"

def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def parse_counts(output: str) -> tuple[int, int, int, bool]:
    import re as _re

    summaries = _re.findall(r"=+\s*(.*?)\s+in\s+[0-9.]+s\s*=+", output)
    if not summaries:
        return 0, 0, 0, False
    final = summaries[-1]
    def count(label: str) -> int:
        match = _re.search(rf"(\d+)\s+{label}", final)
        return int(match.group(1)) if match else 0
    return count("passed"), count("failed"), count("skipped"), True


def run_case(method: str, solution_patch: Path, test_patch: Path) -> dict[str, object]:
    log_path = OUT / f"{method}.docker.log"
    solution_step = "" if method == "baseline" else "git apply --ignore-whitespace --verbose /tmp/solution.diff; "
    shell = "".join([
        "set -u; cd /testbed; git config --global --add safe.directory /testbed; ",
        f"git reset --hard {BASE_COMMIT}; git clean -fdx; ",
        "printf 'HEAD='; git rev-parse HEAD; printf 'DIRTY='; git status --porcelain | wc -l; ",
        solution_step,
        "git apply --ignore-whitespace --verbose /tmp/test.diff; ",
        "python -m pip install -e . --no-build-isolation; ",
        "/opt/miniconda3/envs/testbed/bin/pytest -rA lib/matplotlib/tests/test_figure.py",
    ])
    command = [
        "docker", "run", "--rm", "--entrypoint", "/bin/bash",
        "--mount", f"type=bind,source={solution_patch},target=/tmp/solution.diff,readonly",
        "--mount", f"type=bind,source={test_patch},target=/tmp/test.diff,readonly",
        IMAGE, "-lc", shell,
    ]
    completed = subprocess.run(
        command, cwd=ROOT, text=True, stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT, encoding="utf-8", errors="replace",
    )
    log_path.write_text(completed.stdout, encoding="utf-8")
    passed, failed, skipped, test_executed = parse_counts(completed.stdout)
    patch_count = completed.stdout.count("Applied patch")
    solution_applied = method != "baseline" and patch_count >= 2
    test_applied = patch_count >= 1
    reset_ok = f"HEAD={BASE_COMMIT}" in completed.stdout and "DIRTY=0" in completed.stdout
    return {
        "task_id": TASK_ID,
        "method": method,
        "image": IMAGE,
        "image_id": IMAGE_ID,
        "base_commit": BASE_COMMIT,
        "solution_patch_sha256": sha256(solution_patch),
        "test_patch_sha256": sha256(test_patch),
        "reset_ok": reset_ok,
        "solution_patch_applied": solution_applied,
        "test_patch_applied": test_applied,
        "test_executed": test_executed,
        "pytest_passed": passed,
        "pytest_failed": failed,
        "pytest_skipped": skipped,
        "returncode": completed.returncode,
        "resolved_process_signal": bool(method != "baseline" and reset_ok and solution_applied and test_applied and test_executed and failed == 0 and completed.returncode == 0),
        "formal_outcome": False,
        "provider_calls_added": 0,
    }


def main() -> int:
    if OUT.exists() or ZIP.exists():
        raise SystemExit(f"refusing_to_overwrite:{OUT}")
    if sha256(OBJECTIVE) != ACTIVE_SHA:
        raise SystemExit("active_objective_sha_mismatch")
    for path in (GOLD_PATCH, TEST_PATCH, ORCA_PATCH):
        if not path.is_file():
            raise SystemExit(f"missing_input:{path}")
    OUT.mkdir(parents=True)
    (OUT / "baseline.patch").write_text("", encoding="utf-8", newline="\n")
    (OUT / "gold.patch").write_text(GOLD_PATCH.read_text(encoding="utf-8"), encoding="utf-8", newline="\n")
    (OUT / "test.patch").write_text(TEST_PATCH.read_text(encoding="utf-8"), encoding="utf-8", newline="\n")
    (OUT / "orcaloca.patch").write_text(ORCA_PATCH.read_text(encoding="utf-8"), encoding="utf-8", newline="\n")
    results = [
        run_case("baseline", OUT / "baseline.patch", OUT / "test.patch"),
        run_case("gold", OUT / "gold.patch", OUT / "test.patch"),
        run_case("orcaloca", OUT / "orcaloca.patch", OUT / "test.patch"),
    ]
    summary = {
        "analysis_id": OUT.name,
        "generated_at_utc": datetime.now(timezone.utc).isoformat(),
        "active_objective_sha256": ACTIVE_SHA,
        "task_id": TASK_ID,
        "repository": "matplotlib/matplotlib",
        "image": IMAGE,
        "image_id": IMAGE_ID,
        "base_commit": BASE_COMMIT,
        "test_scope": "lib/matplotlib/tests/test_figure.py",
        "result_rows": len(results),
        "baseline_passed": int(results[0]["pytest_passed"]),
        "baseline_failed": int(results[0]["pytest_failed"]),
        "baseline_skipped": int(results[0]["pytest_skipped"]),
        "gold_f2p": int(results[1]["pytest_passed"]) - int(results[0]["pytest_passed"]),
        "orcaloca_f2p": int(results[2]["pytest_passed"]) - int(results[0]["pytest_passed"]),
        "gold_passed": int(results[1]["pytest_passed"]),
        "orcaloca_passed": int(results[2]["pytest_passed"]),
        "fixed_failed": int(results[1]["pytest_failed"]) + int(results[2]["pytest_failed"]),
        "test_executed_rows": sum(bool(row["test_executed"]) for row in results),
        "resolved_process_signal_rows": sum(bool(row["resolved_process_signal"]) for row in results),
        "pytest_passed": sum(int(row["pytest_passed"]) for row in results),
        "pytest_failed": sum(int(row["pytest_failed"]) for row in results),
        "pytest_skipped": sum(int(row["pytest_skipped"]) for row in results),
        "provider_calls_added": 0,
        "formal_outcome_rows": 0,
        "formal_promotion": False,
        "status": "completed_exact_base_three_condition_replay" if all(bool(row["test_executed"]) for row in results) else "partial_exact_base_three_condition_replay",
        "claim_boundary": "Cached task-image baseline/gold/OrcaLoca process comparison after reset to the declared base commit. F2P/P2P are test-sensitivity diagnostics, not semantic, robust, causal, survival, or formal outcomes.",
        "results": results,
    }
    fields = list(dict.fromkeys(key for row in results for key in row))
    with (OUT / "result_rows.csv").open("w", encoding="utf-8", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=fields)
        writer.writeheader()
        writer.writerows(results)
    (OUT / "summary.json").write_text(json.dumps(summary, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    (OUT / "RESULTS_ZH.md").write_text(
        "# Matplotlib-23476 exact-base three-condition replay\n\n"
        f"baseline={summary['baseline_passed']} passed/{summary['baseline_failed']} failed/{summary['baseline_skipped']} skipped；"
        f"gold={summary['gold_passed']} passed；OrcaLoca={summary['orcaloca_passed']} passed；"
        f"F2P gold/OrcaLoca={summary['gold_f2p']}/{summary['orcaloca_f2p']}。\n\n"
        "该包是缓存 task image 中的 exact-base process/test sensitivity diagnostic，不是 formal outcome。\n",
        encoding="utf-8",
    )
    files = sorted(path for path in OUT.rglob("*") if path.is_file())
    manifest = [{"path": str(path.relative_to(OUT)).replace("\\", "/"), "bytes": path.stat().st_size, "sha256": sha256(path)} for path in files]
    (OUT / "file_manifest.json").write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    files = sorted(path for path in OUT.rglob("*") if path.is_file())
    with zipfile.ZipFile(ZIP, "w", compression=zipfile.ZIP_DEFLATED) as archive:
        for path in files:
            archive.write(path, path.relative_to(OUT))
    with zipfile.ZipFile(ZIP) as archive:
        if archive.testzip() is not None:
            raise SystemExit("archive_integrity_failed")
    print(json.dumps(summary, ensure_ascii=False))
    return 0 if summary["status"] == "completed_exact_base_three_condition_replay" else 1


if __name__ == "__main__":
    raise SystemExit(main())
