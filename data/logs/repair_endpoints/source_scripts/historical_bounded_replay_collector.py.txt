from __future__ import annotations

import csv
import hashlib
import json
import shutil
import zipfile
from datetime import datetime, timezone
from pathlib import Path


ROOT = Path(__file__).resolve().parents[1]
OUT = ROOT / "artifacts/work2_three_more_final_replay_20260907_v1"
ZIP = OUT.with_suffix(".zip")
COMPAT = ROOT / "artifacts/work2_sphinx_pylint_compat_replay_20260907_v3"
MATPLOT = ROOT / "artifacts/work2_three_more_replay_20260907_v2"
ACTIVE_SHA = "390f6f7e27993fe039dc2fc278d840b6870ac939c5bacca3232960f9b7e273e9"


def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def main() -> int:
    if OUT.exists() or ZIP.exists():
        raise SystemExit(f"refusing_to_overwrite:{OUT}")
    OUT.mkdir(parents=True)
    excluded = {"summary.json", "result_rows.csv", "file_manifest.json"}
    for source in sorted(COMPAT.iterdir()):
        if source.is_file() and source.name not in excluded:
            shutil.copy2(source, OUT / source.name)
    selected = "matplotlib__matplotlib-23314."
    for source in sorted(MATPLOT.iterdir()):
        if source.is_file() and source.name.startswith(selected):
            shutil.copy2(source, OUT / source.name)

    compat_summary = json.loads((COMPAT / "summary.json").read_text(encoding="utf-8"))
    matplot_summary = json.loads((MATPLOT / "summary.json").read_text(encoding="utf-8"))
    rows = list(csv.DictReader((COMPAT / "result_rows.csv").open(encoding="utf-8", newline="")))
    rows.extend(row for row in csv.DictReader((MATPLOT / "result_rows.csv").open(encoding="utf-8", newline="")) if row["task_id"] == "matplotlib__matplotlib-23314")
    fields = list(dict.fromkeys(key for row in rows for key in row))
    with (OUT / "result_rows.csv").open("w", encoding="utf-8", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=fields)
        writer.writeheader()
        writer.writerows(rows)
    provenance = compat_summary["task_provenance"] + [
        item for item in matplot_summary["task_provenance"] if item["task_id"] == "matplotlib__matplotlib-23314"
    ]
    task_names = ", ".join(item["task_id"] for item in provenance)
    summary = {
        "analysis_id": OUT.name,
        "generated_at_utc": datetime.now(timezone.utc).isoformat(),
        "active_objective_sha256": ACTIVE_SHA,
        "task_count": len(provenance),
        "result_rows": len(rows),
        "test_executed_rows": sum(row["test_executed"] == "True" for row in rows),
        "resolved_process_signal_rows": sum(row["resolved_process_signal"] == "True" for row in rows),
        "formal_outcome_rows": 0,
        "provider_calls_added": 0,
        "formal_promotion": False,
        "status": "completed_three_more_final_replay" if all(row["test_executed"] == "True" for row in rows) else "partial_three_more_final_replay",
        "claim_boundary": f"Curated exact-base Linux container process/test diagnostics for {task_names}; F2P is a test-sensitivity diagnostic, not semantic, robust, causal, survival, or formal outcome.",
        "source_packages": [str(COMPAT), str(MATPLOT)],
        "task_provenance": provenance,
        "results": rows,
    }
    (OUT / "summary.json").write_text(json.dumps(summary, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    (OUT / "RESULTS_ZH.md").write_text(
        f"# {task_names} 三条件回放\n\n"
        "该包去重整合 Sphinx/Pylint 兼容依赖重跑与 Matplotlib-23314 完整回放；保留 Docker 日志、补丁、测试输出和来源 hash。\n\n"
        "不产生 provider 调用或 formal outcome。\n",
        encoding="utf-8",
    )
    files = sorted(path for path in OUT.rglob("*") if path.is_file())
    (OUT / "file_manifest.json").write_text(json.dumps([{"path": str(path.relative_to(OUT)).replace("\\", "/"), "bytes": path.stat().st_size, "sha256": sha256(path)} for path in files], ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    files = sorted(path for path in OUT.rglob("*") if path.is_file())
    with zipfile.ZipFile(ZIP, "w", compression=zipfile.ZIP_DEFLATED) as archive:
        for path in files:
            archive.write(path, path.relative_to(OUT))
    with zipfile.ZipFile(ZIP) as archive:
        if archive.testzip() is not None:
            raise SystemExit("archive_integrity_failed")
    print(json.dumps(summary, ensure_ascii=False))
    return 0 if summary["status"] == "completed_three_more_final_replay" else 1


if __name__ == "__main__":
    raise SystemExit(main())
